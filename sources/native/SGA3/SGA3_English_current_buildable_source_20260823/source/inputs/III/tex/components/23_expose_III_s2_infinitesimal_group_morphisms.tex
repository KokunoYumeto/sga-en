

\section*{\SGAInlineRefTarget{sga3:iii:section:2}2. Infinitesimal extensions of a morphism of group schemes}
\addcontentsline{toc}{section}{2. Infinitesimal extensions of a morphism of group schemes}

Retain the notation of \S0. Let \(Y\) and \(X\) be two \(S\)-group
functors, and let \(M\) be the kernel of the group homomorphism
\(p_X:X\to X^+\). Thus there is an exact sequence of \(S\)-group
functors
\[
 1\longrightarrow M\longrightarrow X\xrightarrow{p_X}X^+.
\]
By the definition of \(X^+\), there are isomorphisms
\begin{align*}
 \operatorname{Hom}_S(Y,X^+)
  &\xrightarrow{\sim}\operatorname{Hom}_{S_{\mathscr J}}(Y_{\mathscr J},X_{\mathscr J}),\\
 \operatorname{Hom}_{S\text{-}\mathrm{gr}}(Y,X^+)
  &\xrightarrow{\sim}
    \operatorname{Hom}_{S_{\mathscr J}\text{-}\mathrm{gr}}
    (Y_{\mathscr J},X_{\mathscr J}).
\end{align*}
The map
\[
 \operatorname{Hom}_S(Y,p_X):
 \operatorname{Hom}_S(Y,X)\longrightarrow\operatorname{Hom}_S(Y,X^+)
\]
sends an \(S\)-morphism \(f:Y\to X\) to the \(S\)-morphism
\(f^+:Y\to X^+\) corresponding under these isomorphisms to the
\(S_{\mathscr J}\)-morphism
\(f_{\mathscr J}:Y_{\mathscr J}\to X_{\mathscr J}\) obtained from \(f\)
by base change. If \(M\) is commutative, \SGARefLink{sga3:iii:proposition:1-2-4}{Proposition 1.2.4} applies to
this situation.

In practice, we shall be interested in the following case: \(Y\) is
\emph{flat} over \(S\), and \(X\) is either representable (case (a)) or of the
form \(\underline{\operatorname{Aut}}_S(E)\) where \(E\) is representable (case
(b)). The corresponding \(S\)-group functors \(M\) were computed in
\SGARefLink{sga3:iii:corollary:0-9}{0.9} (resp. \SGARefLink{sga3:iii:proposition:0-11}{0.11}), the actions of the inner automorphisms of \(X\) on
\(M\) in \SGARefLink{sga3:iii:corollary:0-10}{0.10} (resp. \SGARefLink{sga3:iii:corollary:0-12}{0.12}). On the other hand, \(Y\) being assumed flat,
by \SGARefLink{sga3:iii:lemma:1-1-1}{1.1.1} it suffices to know the values of \(M\) on the \(S\)-schemes
\emph{flat} over \(S\), and the action of \(X\) on \(M\) when \(X\) and \(M\)
are regarded as functors on the category of \(S\)-schemes \emph{flat} over
\(S\). Finally, in case (a), \(M\) is (on this category) of the form
\(\prod_{S_0/S}N\), where \(N\) is a certain \(S_0\)-functor in abelian
groups (\SGARefLink{sga3:iii:remark:0-5}{0.5}); one will therefore have, by \SGARefLink{sga3:iii:lemma:1-1-2}{1.1.2}, isomorphisms
\[
 H^i(Y,M)\xrightarrow{\ \sim\ }H^i(Y_0,N).
\]
Applying \SGARefLink{sga3:iii:proposition:1-2-4}{1.2.4}, one obtains the

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:theorem:2-1}\textbf{Theorem 2.1.}
\textit{Let \(S\) be a scheme, and let \(\mathscr I\) and \(\mathscr J\) be
two quasi-coherent ideals such that \(\mathscr I\supset\mathscr J\) and
\(\mathscr I\cdot\mathscr J=0\), defining the closed subschemes \(S_0\) and
\(S_{\mathscr J}\). For every \(S\)-functor \(Z\), let \(Z_0\) and
\(Z_{\mathscr J}\) be the functors obtained by base change. Let \(X\) be an
\(S\)-group functor of one of the following two kinds:}
\begin{enumerate}
\item[\(a\))] \textit{\(X\) is an \(S\)-group scheme;}
\item[\(b\))] \textit{\(X=\underline{\operatorname{Aut}}_S(E)\), where \(E\) is
  an \(S\)-scheme of finite presentation over \(S\).}
\end{enumerate}
\textit{Let}
\begin{enumerate}
\item[\(a\))] \textit{\(L_0\) be the \(S_0\)-functor in commutative groups
  defined by}
  \[
   \operatorname{Hom}_{S_0}(T,L_0)
   =\operatorname{Hom}_{\mathscr O_T}
   \bigl(\omega^1_{X_0/S_0}\otimes_{\mathscr O_{S_0}}\mathscr O_T,
   \mathscr J\otimes_{\mathscr O_{S_0}}\mathscr O_T\bigr),
  \]
  \textit{on which \(X_0\) acts via its adjoint representation in
  \(\omega^1_{X_0/S_0}\);}
\item[\(b\))] \textit{\(L\) be the functor in abelian groups on the category of
  \(S\)-schemes flat over \(S\) defined by}
  \[
   \operatorname{Hom}_S(T,L)
   =\Gamma\bigl(T,f_*\bigl(\mathscr{H}om_{\mathscr O_E}
   (\Omega^1_{E/S},\mathscr J\mathscr O_E)\bigr)
   \otimes_{\mathscr O_S}\mathscr O_T\bigr)
  \]
  \textit{(\(f\) denotes the structural morphism \(E\to S\)), on which \(X\),
  regarded as a functor on the same category, acts as was seen in \SGARefLink{sga3:iii:corollary:0-12}{0.12}.}
\end{enumerate}
\textit{Let \(Y\) be an \(S\)-group scheme flat over \(S\), and
\(f_{\mathscr J}:Y_{\mathscr J}\to X_{\mathscr J}\) a morphism of
\(S_{\mathscr J}\)-groups. In case (a), let \(Y_0\) act on \(L_0\) through
\(f_0:Y_0\to X_0\). Then:}
\begin{enumerate}
\item[(i)] \textit{For \(f_{\mathscr J}\) to lift to a morphism of \(S\)-groups
  \(Y\to X\), it is necessary and sufficient that the following two conditions
  be satisfied:}
  \begin{enumerate}
  \item[(i\(_1\))] \textit{\(f_{\mathscr J}\) lifts to a morphism of
    \(S\)-functors \(Y\to X\), which in case (b) allows one to let \(Y\) act on
    \(L\) through the lifting \(Y\to X\) (an action which, as we have seen, is
    independent of the lifting chosen);}
  \item[(i\(_2\))] \textit{a certain obstruction \(c(f_{\mathscr J})\),
    canonically defined by \(f_{\mathscr J}\), vanishes, where
    \(c(f_{\mathscr J})\) is a class in}
    \[
     (a)\ H^2(Y_0,L_0),\qquad (b)\ H^2(Y,L).
    \]
  \end{enumerate}
\item[\SGAInlineRefTarget{sga3:iii:theorem:2-1:item:ii}(ii)] \textit{If the conditions of (i) are satisfied, the set
  of morphisms of \(S\)-group functors \(Y\to X\) extending
  \(f_{\mathscr J}\), modulo the action of the inner automorphisms of \(X\)
  defined by the sections of \(X\) over \(S\) inducing the unit section of
  \(X_{\mathscr J}\) over \(S_{\mathscr J}\), is principal homogeneous under}
  \[
   (a)\ H^1(Y_0,L_0),\qquad (b)\ H^1(Y,L).
  \]
\item[\SGAInlineRefTarget{sga3:iii:theorem:2-1:item:iii}(iii)] \textit{If \(f:Y\to X\) is a morphism of
  \(S\)-group functors extending \(f_{\mathscr J}\), the set of morphisms
  \(Y\to X\) transformed from \(f\) by the inner automorphisms defined by the
  sections of \(X\) over \(S\) inducing the unit section of
  \(X_{\mathscr J}\) over \(S_{\mathscr J}\) is isomorphic to}
  \[
   (a)\ \Gamma(L_0)/H^0(Y_0,L_0),\qquad (b)\ \Gamma(L)/H^0(Y,L).
  \]
\end{enumerate}

\emph{Addition} (after the Gille--Polo re-edition (2011), Exposé~III; our
formulation). The re-edition also states, in (ii), that the set
\(\mathcal E\) itself of morphisms of \(S\)-group functors \(Y\to X\) extending
\(f_{\mathscr J}\) is principal homogeneous under \(Z^1(Y,L)\) (under
\(Z^1(Y_0,L_0)\) in case (a)); this is 1.2.2 (ii), applied to the extension
\((E_f)\) of 1.2.3, as in the proof of 1.2.4. It adds a Remark 2.1.1 fixing
notation: if \(f,f'\in\mathcal E\), let \(d(f,f')\) be the cocycle such that
\(f'=d(f,f')\cdot f\), and \(\bar d(f,f')\) its class in \(H^1(Y,L)\) (in
\(H^1(Y_0,L_0)\) in case (a)); by (ii), \(\bar d(f,f')=0\) exactly when \(f\)
and \(f'\) are transformed from one another by an inner automorphism of the
kind considered there. It also notes, in (i\(_1\)), that in case (a) the
action of \(Y\) on \(L\) given by the lifting comes from \(f_0\) and the
adjoint action of \(X_0\) on \(L_0\), as stated before (i).

\noindent\SGAInlineRefTarget{sga3:iii:remark:2-2}\textbf{Remark 2.2.}
Suppose moreover in (a) that \(X\) is locally of finite presentation over
\(S\). Applying 0.6, one then obtains
\begin{enumerate}
\item[a)] \(H^i(Y_0,L_0)=H^i\bigl(Y_0,\mathscr{H}om_{\mathscr O_{S_0}}
  (\omega^1_{X_0/S_0},\mathscr J)\bigr)\),
\item[b)] \(H^i(Y,L)=H^i\bigl(Y,f_*\bigl(\mathscr{H}om_{\mathscr O_E}
  (\Omega^1_{E/S},\mathscr J\mathscr O_E)\bigr)\bigr)\),
\end{enumerate}
the modules whose cohomology is taken being
\emph{quasi-coherent}.\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~III, adds here the hypothesis that
\(Y\) is affine over \(S\), with a reference to
\SGARefLink{sga3:i:subsection:5-3}{Exposé~I, 5.3}, where, for an affine group
and a quasi-coherent module \(\mathscr F\), \(H^i(Y,\mathscr F)\) is defined as
\(H^i(Y,\mathbf W(\mathscr F))\) and computed by the complex
\(\mathscr F\otimes\mathscr A(Y)^{\otimes\bullet}\). With \(H^i(Y,\mathscr F)\)
read as \(H^i(Y,\mathbf W(\mathscr F))\), the formulas hold as printed: \(L_0\)
coincides with \(\mathbf W(\mathscr{H}om(\omega^1_{X_0/S_0},\mathscr J))\) on
the \(S_0\)-schemes flat over \(S_0\) (0.6), \(L\) with
\(\mathbf W(f_*\mathscr{H}om(\Omega^1_{E/S},\mathscr J\mathscr O_E))\) by its
definition, and the powers of \(Y_0\) (resp. \(Y\)) are flat over \(S_0\)
(resp. \(S\)), so that 1.1.1 applies.}

Moreover, under these conditions, one may write in case
(a)\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~III, adds the analogue in case (b). In our words: \(E\),
being smooth, is flat over \(S\), so by the Addition to 0.14 the functor \(L\)
is \(\prod_{S_0/S}\mathbf W\bigl(\pi_{0*}\mathscr{H}om_{\mathscr O_{E_0}}
(\Omega^1_{E_0/S_0},\mathscr J\otimes_{\mathscr O_{S_0}}\mathscr O_{E_0})\bigr)\)
on the \(S\)-schemes flat over \(S\), where \(\pi_0\colon E_0\to S_0\) is the
structural morphism; by 1.1.2, \(H^i(Y,L)=H^i\bigl(Y_0,\pi_{0*}\mathscr{H}om
_{\mathscr O_{E_0}}(\Omega^1_{E_0/S_0},\mathscr J\otimes_{\mathscr O_{S_0}}
\mathscr O_{E_0})\bigr)\).}
\[
 H^i(Y_0,L_0)=H^i\bigl(Y_0,\underline{\operatorname{Lie}}(X_0/S_0)
 \otimes_{\mathscr O_{S_0}}\mathscr J\bigr).
\]

Let us now state a number of corollaries concerning the case where \(Y\)
is a \emph{diagonalizable} group (I, \SGARefLink{sga3:i:subsection:4-4}{4.4}); one then knows (loc.\ cit.\
\SGARefLink{sga3:i:theorem:5-3-3}{5.3.3}) that, if \(S\) is affine, the cohomology of \(Y\) with values in
every quasi-coherent \(\mathscr O_S\)-module is zero. First a special case:

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:corollary:2-4}\textbf{Corollary 2.4.}
\textit{Let \(S_0\hookrightarrow S\) be a closed subscheme defined by
a nilpotent ideal, and let \(Y\) be a diagonalizable \(S\)-group. Let}
\begin{enumerate}
\item[\(a\)] \textit{\(X\) be an \(S\)-group locally of finite
  presentation over \(S\), or}
\item[\(b\)]\SGAInlineRefTarget{sga3:iii:corollary:2-4:item:b} \textit{\(X=\underline{\operatorname{Aut}}_S(E)\), where
  \(E\) is an \(S\)-scheme locally of finite presentation.}
\end{enumerate}
\textit{If \(f:Y\to X\) is a morphism of \(S\)-groups whose base
change \(f_0:Y_0\to X_0\) is the unit morphism, then \(f\) is the unit
morphism.}

Indeed, the question is local on \(S\), and in case \(b\) also local on
\(E\). We may therefore suppose that \(S\) is affine and, in case \(b\),
that \(E\) is of finite presentation over \(S\). Introduce the closed
subschemes \(S_n\subset S\) defined by the powers of the ideal defining
\(S_0\). This reduces the assertion to the case in which \(S_0\) is
defined by a square-zero ideal, where it follows from \SGARefLink{sga3:iii:theorem:2-1}{Theorem 2.1} and
\SGARefLink{sga3:iii:remark:2-2}{Remark 2.2}.

If \(f_0\) is not required to be the unit morphism, one has the
following statement.

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:corollary:2-5}\textbf{Corollary 2.5.}
\textit{Let \(S\) and \(S_0\) be as in \SGARefLink{sga3:iii:corollary:2-4}{2.4}; suppose moreover \(S\)
affine. Let \(Y\) be a diagonalizable \(S\)-group, \(X\) an \(S\)-group functor,
and \(f_0:Y_0\to X_0\) a morphism of \(S_0\)-group functors.}
\begin{enumerate}
\item[(i)] \textit{Suppose that one of the following two properties holds:}
  \begin{enumerate}
  \item[(a)] \textit{\(X\) is an \(S\)-group locally of finite presentation over
    \(S\);}
  \item[(b)] \textit{\(X=\underline{\operatorname{Aut}}_S(E)\), where \(E\) is of
    finite presentation over \(S\).}
  \end{enumerate}
  \textit{Then \(f_0\) extends to a morphism of \(S\)-groups \(Y\to X\) if and
  only if it extends to a morphism of \(S\)-functors \(Y\to X\).}
\item[\SGAInlineRefTarget{sga3:iii:corollary:2-5:item:ii}(ii)] \textit{Suppose that one of the following two properties holds:}
  \begin{enumerate}
  \item[(a)] \textit{\(X\) is an \(S\)-group smooth over \(S\);}
  \item[(b)] \textit{\(X=\underline{\operatorname{Aut}}_S(E)\), where \(E\) is
    smooth and affine over \(S\).}
  \end{enumerate}
  \textit{Then \(f_0\) extends to a morphism of \(S\)-groups \(Y\to X\), and two
  such extensions are conjugate by an inner automorphism of \(X\) defined by a
  section of \(X\) over \(S\) inducing the unit section of \(X_0\) over
  \(S_0\).}
\end{enumerate}

Introducing the \(S_n\) as above, (i) follows step by step from part (ii) of
the theorem.\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~III, removes assertion (i) from the statement,
quoting it in a note together with this sentence, which it finds insufficient,
and observes that (i) is not used later, where \(X\) is always smooth over \(S\).
In our words: the step from \(S_n\) to \(S_{n+1}\) applies 2.1 (i) to a
morphism of groups \(g_n\colon Y_{S_n}\to X_{S_n}\) lifting \(f_0\), and needs
condition (i\(_1\)) for \(g_n\); the hypothesis provides a morphism of
functors \(Y\to X\) lifting \(f_0\), hence condition (i\(_1\)) for its
restriction to \(S_n\), but this restriction need not be a morphism of
groups, nor equal to \(g_n\) (for instance when \(\mathscr I^3=0\) and
\(n=1\)). The argument thus leaves a gap in (i); (ii) is not affected.} For
(ii), one argues in the same way, noting that the conditions of 2.1 (i) are
automatically satisfied by 2.3 (note that a scheme smooth over \(S\) is
necessarily locally of finite presentation over \(S\), so that a scheme smooth
and affine over \(S\) is necessarily of finite presentation); moreover, every
section of \(X_{S_n}\) over \(S_n\) lifts to a section of \(X_{S_{n+1}}\) over
\(S_{n+1}\) (by the definition of ``smooth over \(S\)'' for (a), and by
\SGARefLink{sga3:iii:corollary:0-17}{0.17} for (b));\footnote{Translator's note: the 1970 text refers to 0.16,
which lifts such a section to an endomorphism of \(E_{S_{n+1}}\); the
Gille--Polo re-edition (2011), Exposé~III, refers to 0.17, which gives an
automorphism (by 0.7).} if \(f\) and \(f'\) are two liftings of \(f_0\), one
may suppose step by step that \(f_n=f'_n\) by lifting the inner automorphism
whose existence is asserted by part (ii) of the theorem, which completes the
proof.

Arguing in the same way, one obtains, for an arbitrary \(Y\):

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:corollary:2-6}\textbf{Corollary 2.6.}
\textit{Let \(\mathscr I\) be a nilpotent ideal of \(S\) defining
\(S_0\), let \(Y\) be an affine \(S\)-group flat over \(S\), and let
\(X\) be a smooth \(S\)-group.}
\begin{enumerate}
\item[\(i\)] \textit{If, for every \(n\geq0\),}
 \[
  H^2\!\left(Y_0,
   \operatorname{Lie}(X_0/S_0)\otimes_{\mathscr O_{S_0}}
   \mathscr I^{n+1}/\mathscr I^{n+2}\right)=0,
 \]
 \textit{then every \(S_0\)-group morphism \(f_0:Y_0\to X_0\) lifts
 to an \(S\)-group morphism \(f:Y\to X\).}
\item[(ii)] \textit{If, for every \(n\geq0\),}
 \[
  H^1\!\left(Y_0,
   \operatorname{Lie}(X_0/S_0)\otimes_{\mathscr O_{S_0}}
   \mathscr I^{n+1}/\mathscr I^{n+2}\right)=0,
 \]
 \textit{then any two such lifts are conjugate by an inner
 automorphism of \(X\) defined by a section over \(S\) inducing the
 unit section of \(X_0\) over \(S_0\).}
\end{enumerate}

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:lemma:2-7}\textbf{Lemma 2.7.}
\textit{Let \(S\) be affine, \(G\) an affine \(S\)-group,
\(\mathscr F\) a quasi-coherent \(\mathscr O_S\)-module, and
\(\mathscr L\) a locally free \(\mathscr O_S\)-module. Suppose \(G\)
acts on \(\mathscr F\) in the sense of Expos\'e I, hence acts on
\(\mathscr F\otimes_{\mathscr O_S}\mathscr L\).\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~III, specifies
that \(G\) acts on \(\mathscr F\otimes_{\mathscr O_S}\mathscr L\) through
\(\mathscr F\), with the trivial action on \(\mathscr L\).} Let \(\Lambda\) be
the ring of \(S\), and let \(L\) be the projective \(\Lambda\)-module
defining \(\mathscr L\). Then there is a canonical isomorphism}
\[
 H^*(G,\mathscr F\otimes_{\mathscr O_S}\mathscr L)
 \simeq H^*(G,\mathscr F)\otimes_\Lambda L.
\]

Indeed, in the situation of the statement, the cohomology groups of \(G\)
with values in \(\mathscr F\) are computed as the groups of sections of the
cohomology sheaves \(\underline H^n(G,\mathscr F)\) (I, \SGARefLink{sga3:i:subsection:5-3}{5.3}). Now it follows
immediately from the definition of these cohomology sheaves that their
formation commutes with tensor product by a locally free sheaf. Taking
sections (we are over an affine base), one finds the asserted
result.\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~III, replaces this by a direct argument. In our
words: by I, 5.3, \(H^*(G,\mathscr F)\) and
\(H^*(G,\mathscr F\otimes\mathscr L)\) are the cohomology of
\(\Gamma(S,\mathscr C)\) and \(\Gamma(S,\mathscr C\otimes\mathscr L)\), where
\(\mathscr C\) is the complex \(\mathscr F\to\mathscr F\otimes\mathscr A(G)
\to\mathscr F\otimes\mathscr A(G)\otimes\mathscr A(G)\to\cdots\); since \(S\) is
affine, \(\Gamma(S,\mathscr C\otimes\mathscr L)=\Gamma(S,\mathscr C)
\otimes_\Lambda L\) (EGA~I, 1.3.12), and \(L\), being projective, is flat over
\(\Lambda\), so that taking cohomology commutes with \(\otimes_\Lambda L\).}

Using the lemma, \SGARefLink{sga3:iii:corollary:2-6}{Corollary 2.6} becomes:

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:corollary:2-8}\textbf{Corollary 2.8.}
\textit{Let \(S\) be affine and let \(\mathscr I\) be a nilpotent
ideal defining \(S_0\). Suppose every
\(\mathscr I^{n+1}/\mathscr I^{n+2}\) is locally free on \(S_0\). Let
\(Y\) be an affine \(S\)-group flat over \(S\), let \(X\) be a smooth
\(S\)-group, and let \(f_0:Y_0\to X_0\) be an \(S_0\)-group morphism.}
\begin{enumerate}
\item[\(i\)]\SGAInlineRefTarget{sga3:iii:corollary:2-8:item:i} \textit{If
\(H^2(Y_0,\operatorname{Lie}(X_0/S_0))=0\), then \(f_0\) lifts to an
\(S\)-group morphism \(Y\to X\).}
\item[\SGAInlineRefTarget{sga3:iii:corollary:2-8:item:ii}(ii)] \textit{If
\(H^1(Y_0,\operatorname{Lie}(X_0/S_0))=0\), then any two such lifts are
conjugate by an inner automorphism of \(X\) defined by a section over
\(S\) inducing the unit section of \(X_0\) over \(S_0\).}
\end{enumerate}

Taking \(Y=X\) gives:

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:corollary:2-9}\textbf{Corollary 2.9.}
\textit{Let \(S\) and \(S_0\) be as above, and let \(X\) be a smooth
affine \(S\)-group.}
\begin{enumerate}
\item[\(i\)] \textit{If
\(H^1(X_0,\operatorname{Lie}(X_0/S_0))=0\), every endomorphism of \(X\)
over \(S\) that induces the identity on \(X_0\) is the inner
automorphism defined by a section over \(S\) inducing the unit section
of \(X_0\) over \(S_0\).}
\item[(ii)] \textit{If
\(H^2(X_0,\operatorname{Lie}(X_0/S_0))=0\), every
\(S_0\)-automorphism of \(X_0\) extends to an \(S\)-automorphism of
\(X\).}\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~III, gives the argument for (ii). In our words: by
2.8 (i), applied with \(Y=X\), an \(S_0\)-automorphism \(f_0\) of \(X_0\) and
its inverse lift to \(S\)-endomorphisms \(f\) and \(g\) of \(X\); then
\(g\circ f\) and \(f\circ g\) induce the identity on \(X_0\), hence are
automorphisms (0.7, applied to the successive \(S_n\)), and so is \(f\).}
\end{enumerate}

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:remark:2-10}\textbf{Remark 2.10.}
The assertions involving \(H^1\) have converses by \SGARefLink{sga3:iii:theorem:2-1}{Theorem 2.1}. For
example, let \(S=I_{S_0}\) be the dual-number scheme over \(S_0\)
(II, 2.1), and let \(X\) be a flat \(S\)-group such that every
automorphism over \(S\) inducing the identity over \(S_0\) is an inner
automorphism defined by a section over \(S\) inducing the unit section
of \(X_0\). Then
\[
 H^1(X_0,\operatorname{Lie}(X_0/S_0))=0.
\]

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:corollary:2-11}\textbf{Corollary 2.11.}
\textit{Let \(S,\mathscr I,\mathscr J\) be as in \SGARefLink{sga3:iii:theorem:2-1}{2.1}. Let \(Y\) be an
\(S\)-group scheme flat over \(S\), let \(X\) be an \(S\)-group scheme,
and let \(f:Y\to X\) be an \(S\)-group morphism. The set of morphisms
from \(Y\) to \(X\) obtained from \(f\) by conjugation with elements
\(x\in X(S)\) inducing the unit of \(X(S_{\mathscr J})\) is isomorphic
to the quotient}
\[
 \mathcal E=
 \operatorname{Hom}_{\mathscr O_{S_0}}(\omega^1_{X_0/S_0},\mathscr J)
 \big/
 \operatorname{Hom}_{\mathscr O_{S_0}}(\omega^1_{X_0/S_0},\mathscr J)
 ^{\operatorname{ad}(Y_0)}.
\]
\textit{Here the second group consists of those
\(\mathscr O_{S_0}\)-morphisms
\(\omega^1_{X_0/S_0}\to\mathscr J\) which, after every base change
\(S'\to S_0\), yield morphisms}
\[
 \omega^1_{X_0/S_0}\otimes_{\mathscr O_{S_0}}\mathscr O_{S'}
 \longrightarrow
 \mathscr J\otimes_{\mathscr O_{S_0}}\mathscr O_{S'}
\]
\textit{invariant under the action of \(Y_0(S')\) on the first
factor.}

By \SGARefLink{sga3:iii:theorem:2-1:item:iii}{Theorem 2.1(iii)}, this set is
\(\Gamma(L_0)/H^0(Y_0,L_0)\). Now
\[
 \Gamma(L_0)=
 \operatorname{Hom}_{\mathscr O_{S_0}}(\omega^1_{X_0/S_0},\mathscr J),
\]
and \(H^0(Y_0,L_0)\) is precisely
\(\Gamma(L_0)^{\operatorname{ad}(Y_0)}\) in the sense just stated.

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:corollary:2-12}\textbf{Corollary 2.12.}
\textit{Under the hypotheses of \SGARefLink{sga3:iii:corollary:2-11}{2.11}, suppose in addition that
\(\omega^1_{X_0/S_0}\) is locally free of finite rank. Then}\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~III,
gives the reason: for \(\omega^1_{X_0/S_0}\) locally free of finite rank,
\(\operatorname{Hom}_{\mathscr O_{S_0}}(\omega^1_{X_0/S_0},\mathscr J)\simeq
\Gamma\bigl(S_0,\underline{\operatorname{Lie}}(X_0/S_0)\otimes_{\mathscr O_{S_0}}
\mathscr J\bigr)\), \(\underline{\operatorname{Lie}}(X_0/S_0)\) being the dual of
\(\omega^1_{X_0/S_0}\).}
\[
 \mathcal E\simeq
 \Gamma\!\left(S_0,
  \operatorname{Lie}(X_0/S_0)\otimes_{\mathscr O_{S_0}}\mathscr J
 \right)
 \big/
 H^0\!\left(Y_0,
  \operatorname{Lie}(X_0/S_0)\otimes_{\mathscr O_{S_0}}\mathscr J
 \right).
\]

\medskip
\noindent\SGAInlineRefTarget{sga3:iii:corollary:2-13}\textbf{Corollary 2.13.}
\textit{Suppose in addition that \(Y_0\) is diagonalizable. Then}
\[
 \mathcal E\simeq
 \Gamma\!\left(S_0,
  \operatorname{Lie}(X_0/S_0)\otimes_{\mathscr O_{S_0}}\mathscr J
 \right)
 \big/
 \Gamma\!\left(S_0,
  \operatorname{Lie}(X_0/S_0)^{\operatorname{ad}(Y_0)}
  \otimes_{\mathscr O_{S_0}}\mathscr J
 \right),
\]
\textit{where
\(\operatorname{Lie}(X_0/S_0)^{\operatorname{ad}(Y_0)}\) is the factor
of the decomposition in I, \SGARefLink{sga3:i:proposition:4-7-3}{4.7.3} corresponding to the zero
character of \(Y_0\).}

Indeed, if \(Y_0\simeq D_{S_0}(M)\), the cited result gives a direct-sum
decomposition
\[
 \operatorname{Lie}(X_0/S_0)
 =\operatorname{Lie}(X_0/S_0)_0
 \oplus
 \bigoplus_{\substack{m\in M\\m\ne0}}
 \operatorname{Lie}(X_0/S_0)_m.
\]
Tensoring with \(\mathscr J\) gives the analogous decomposition for
\(\operatorname{Lie}(X_0/S_0)\otimes\mathscr J\), whence
\[
 H^0\!\left(Y_0,
  \operatorname{Lie}(X_0/S_0)\otimes\mathscr J\right)
 \simeq
 \Gamma\!\left(S_0,
  \operatorname{Lie}(X_0/S_0)_0
  \otimes_{\mathscr O_{S_0}}\mathscr J\right).
\]

\textit{Suppose in addition that \(S_0\) is affine. Then}

 \mathcal E\simeq
 \Gamma\!\left(
  S_0,
  \left[
   \operatorname{Lie}(X_0/S_0)
   \big/
   \operatorname{Lie}(X_0/S_0)^{\operatorname{ad}(Y_0)}
  \right]
  \otimes_{\mathscr O_{S_0}}\mathscr J
 \right).
\]
