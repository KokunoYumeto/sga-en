\begin{lemma}\label{sga3:XXI:3.4.10}
Under the preceding conditions, the following conditions on an
element \(w\in W\) are equivalent:
\begin{enumerate}
\item[\SGAInlineRefTarget{sga3:XXI:3.4.10:item:i}(i)] \(w\in W_{\Delta'}\).
\item[\SGAInlineRefTarget{sga3:XXI:3.4.10:item:ii}(ii)] For every \(m\in M\), one has \(w(m)-m\in V'\).
\item[\SGAInlineRefTarget{sga3:XXI:3.4.10:item:iii}(iii)] For every \(\alpha\in R\), one has
  \(w(\alpha)-\alpha\in V'\).
\end{enumerate}
\end{lemma}

\begin{proof}
Clearly \SGARefLink{sga3:XXI:3.4.10:item:i}{(i)} implies \SGARefLink{sga3:XXI:3.4.10:item:ii}{(ii)}, and \SGARefLink{sga3:XXI:3.4.10:item:ii}{(ii)} implies \SGARefLink{sga3:XXI:3.4.10:item:iii}{(iii)}. We prove \SGARefLink{sga3:XXI:3.4.10:item:iii}{(iii)}\(\Rightarrow\)\SGARefLink{sga3:XXI:3.4.10:item:i}{(i)}. So let \(w\in W\),
\(w\notin W_{\Delta'}\), with \(w(\alpha)-\alpha\in V'\) for every
\(\alpha\in R\). Write \(w\) as a product of reflections
\(s_{\alpha_0}s_{\alpha_1}\cdots s_{\alpha_n}\), with \(\alpha_i\in\Delta\).
After replacing \(w\) by a partial product of the \(s_{\alpha_i}\), we may
suppose that \(w'=s_{\alpha_1}\cdots s_{\alpha_n}\in W_{\Delta'}\). Then
\[
 w(\alpha)-\alpha
   =w'(\alpha)-\alpha-\bigl(w'^{-1}(\alpha_0^\ast),\alpha\bigr)\alpha_0.
\]
Taking \(\alpha=w'^{-1}(\alpha_0)\), one finds \(2\alpha_0\in V'\), hence
\(\alpha_0\in\Delta'\), and finally \(w=s_{\alpha_0}w'\in W_{\Delta'}\),
contrary to the hypothesis.
\end{proof}

\subsection{Miscellaneous remarks}
\label{sga3:XXI:3.5}

\begin{proposition}\label{sga3:XXI:3.5.1}
Let \(R_+\) be a system of positive roots. Put
\[
 \rho_{R_+}
   =\frac12\sum_{\alpha\in\operatorname{ind}(R_+)}\alpha.
\]
Then
\[
 (\beta^\ast,\rho_{R_+})=1
 \qquad\text{for every }\beta\in\mathscr{S}(R_+).
\]
\end{proposition}

\begin{proof}
One can write
\[
 2\rho_{R_+}
   =\beta+
     \sum_{\substack{\alpha\in\operatorname{ind}(R_+)\\
                     \alpha\ne\beta}}\alpha.
\]
Hence
\[
 s_\beta(2\rho_{R_+})=2\rho_{R_+}-2\beta
\]
by Corollary~\ref{sga3:XXI:3.3.2}.
\end{proof}

\begin{corollary}\label{sga3:XXI:3.5.2}
Put
\[
 \rho_{R_+}^\ast
   =\frac12
     \sum_{\alpha^\ast\in\operatorname{ind}(R_+^\ast)}
     \alpha^\ast.
\]
Then:
\begin{enumerate}
\item[(i)] \((\rho_{R_+}^\ast,\alpha)>0\) for every
  \(\alpha\in R_+\); equivalently,
  \(\rho_{R_+}^\ast\in\mathcal{C}(R_+)\), in the notation of
  Definition~\ref{sga3:XXI:3.6.8}.
\item[(ii)] For every \(\alpha\in R\),\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~XXI, deduces (ii) from
3.1.5 and from Proposition~\ref{sga3:XXI:3.5.1} applied to \(\cR^\ast\). In our
formulation, this argument, and (ii), are valid when \(\cR\) is reduced: then
\(\operatorname{ind}(R_+^\ast)=R_+^\ast\) has \(\mathscr{S}(R_+)^\ast\) as system of
simple roots (3.2.12), so 3.5.1 gives \((\rho_{R_+}^\ast,\alpha)=1\) for
\(\alpha\in\mathscr{S}(R_+)\), and (ii) follows by linearity. In general (ii) fails: for
the root datum of type \(A'_1\) of 2.1.2, with \(R_+=\{\alpha,2\alpha\}\), one has
\(\operatorname{ind}(R_+^\ast)=\{\alpha^\ast/2\}\), \(\rho_{R_+}^\ast=\alpha^\ast/4\) and
\((\rho_{R_+}^\ast,\alpha)=1/2\), whereas \(\operatorname{ord}(\alpha)=1\). The same
argument gives \((\rho_{R_+}^\ast,\alpha)\in\{1,\frac12\}\) for simple \(\alpha\), whence
(i) in all cases.}
  \[
   \operatorname{ord}_{\mathscr{S}(R_+)}(\alpha)
      =(\rho_{R_+}^\ast,\alpha).
  \]
\end{enumerate}
\end{corollary}

\begin{remark}\label{sga3:XXI:3.5.3}
If \(w\in W\), then
\[
 \rho_{w(R_+)}=w(\rho_{R_+}),
 \qquad
 \rho_{w(R_+)}^\ast=w(\rho_{R_+}^\ast).
\]
\end{remark}

\begin{proposition}\label{sga3:XXI:3.5.4}
Let \(\alpha\) and \(\gamma\) be two nonproportional roots, with
\(\alpha\) indivisible. There exist a system of simple roots
containing \(\alpha\) and a root \(\beta\) such that
\[
 \gamma=a\alpha+b\beta,
 \qquad a,b\in\mathbb{N}.
\]
\end{proposition}

\begin{proof}
Choose a basis of the vector space \(\VR(R)\) containing
\(\alpha_1=\alpha\) and \(\alpha_2=\gamma\), and consider the
lexicographic order with respect to this basis. Let \(R_+\) be the set
of roots \(>0\). It is clear that \(R_+\) is a system of positive
roots and that the
least element of \(R_+\) not proportional to \(\alpha\) is simple.\footnote{Translator's
note: the 1970 text says that the two smallest elements of \(R_+\) are simple, the
smallest being \(\alpha\) and the next of the form \(\beta=p\alpha+q\gamma\),
\(0<q\leq1\) (in the present notation), and it ends with ``\(q^{-1}>0\), hence also
\(-q^{-1}p>0\)''. When \(2\alpha\) is a root it comes second and is not simple, and
\(-q^{-1}p\) may be \(0\). The corrected argument is that of the Gille--Polo
re-edition (2011), Exposé~XXI. In our words: the elements of \(R_+\) smaller than
\(\beta\) are \(\alpha\) and possibly \(2\alpha\), so neither \(\alpha\) nor \(\beta\) is a
sum of two elements of \(R_+\), and both are simple by 3.2.14; the coefficients of the
positive root \(\gamma\) with respect to \(\mathscr{S}(R_+)\) lie in \(\mathbb N\) by
3.1.5.}
This element has the form
\[
 \beta=p\alpha+q\gamma,
 \qquad 0<q\le1.
\]
Consequently
\[
 \gamma=q^{-1}\beta-q^{-1}p\alpha.
\]
Since \(q^{-1}>0\), one has
\[
 q^{-1}\in\mathbb{N}^{\ast},
 \qquad
 -q^{-1}p\in\mathbb{N}.
\]
\end{proof}
