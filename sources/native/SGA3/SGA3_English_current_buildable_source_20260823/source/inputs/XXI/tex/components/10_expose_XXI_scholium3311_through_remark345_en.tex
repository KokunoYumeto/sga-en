\begin{scholium}\label{sga3:XXI:3.3.11}
Let \(P'\) be a set of roots satisfying (P 2) and closed. There exist a
system of simple roots \(\Delta\) and a subset \(\Delta'\) of
\(\Delta\) such that
\[
 P'=R\cap\bigl(\mathbb{N}\cdot\Delta
          \,\cup\, -\mathbb{N}\cdot\Delta'\bigr).
\]
If
\[
 R_{\Delta'}=(\mathbb{Z}\cdot\Delta')\cap R,
\]
which is a closed and symmetric subset of \(R\),\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~XXI, inserts
before this scholium a definition: a subset \(R'\) of \(R\) is \emph{symmetric} if
\(-R'=R'\); the 1970 text gives it in 3.4.1 below.} then \(P'\) is the
disjoint union of \(R_{\Delta'}\) and the closed subset of
\(\mathscr{P}(\Delta)\) formed by the
\[
 \alpha\in\mathscr{P}(\Delta)\setminus
           \mathbb{N}\cdot\Delta',
\]
that is, the positive roots whose decomposition with respect to
\(\Delta\) ``contains at least one root of
\(\Delta\setminus\Delta'\)''.
\end{scholium}

\subsection{Closed and symmetric sets of roots}
\label{sga3:XXI:3.4}

\begin{proposition}\label{sga3:XXI:3.4.1}
Let \(R'\subset R\) be a set of roots of the root datum \(\cR\).
Suppose that \(R'\) is closed and symmetric (i.e.\ \(R'=-R'\)). Then
\((M,M^\ast,R',R'^\ast)\) is a root datum; for every system of
positive roots \(R_+\) of \(R\), \(R'_+=R_+\cap R'\) is a system of
positive roots of \(R'\); the Weyl group \(W_{R'}\) of
\((M,M^\ast,R',R'^\ast)\) is the subgroup of \(W(\cR)\) generated by
the \(s_\alpha\), \(\alpha\in R'\).
\end{proposition}

\begin{proof}
The first assertion is immediate from
Corollary~\ref{sga3:XXI:1.2.11}; the second follows from
Corollary~\ref{sga3:XXI:3.3.8}; and the third is evident.
\end{proof}

\begin{corollary}\label{sga3:XXI:3.4.2}
Let \(w\in W_{R'}\). The order of \(w\) is the smallest integer
\(n>0\) such that
\[
 w^n(\alpha')=\alpha'
 \qquad\text{for every }\alpha'\in R'.
\]
\end{corollary}

\begin{proof}
It suffices to apply Corollary~\ref{sga3:XXI:1.2.7} to the root datum
\((M,M^\ast,R',R'^\ast)\).
\end{proof}

\begin{corollary}\label{sga3:XXI:3.4.3}
Let \(\alpha\) and \(\beta\) be two nonproportional roots. Let \(n\)
be the smallest integer \(>0\) such that
\[
 (s_\alpha s_\beta)^n(\alpha)=\alpha,
 \qquad
 (s_\alpha s_\beta)^n(\beta)=\beta.
\]
Then the subgroup \(W_{\alpha,\beta}\) of \(W\) generated by
\(s_\alpha\) and \(s_\beta\) is defined by the relations
\[
 s_\alpha^2=1,\qquad
 s_\beta^2=1,\qquad
 (s_\alpha s_\beta)^n=1.
\]
\end{corollary}

\begin{proof}
In view of Corollary~\ref{sga3:XXI:3.4.2}, it remains only to verify
the following lemma.
\end{proof}

\begin{lemma}\label{sga3:XXI:3.4.4}
Let \(G\) be the group generated by two elements \(x\) and \(y\)
subject to the relations \(x^2=y^2=1\). Every normal subgroup of
\(G\) containing neither \(x\) nor \(y\) is generated, as a normal
subgroup, by an element of the form \((xy)^n\).
\end{lemma}

\begin{proof}
Every element of \(G\) can be written as \((xy)^n\), as
\((yx)^n=(xy)^{-n}\), or in one of the forms
\[
\begin{array}{c@{\qquad}c@{\qquad}c}
 (a) & x(yx)^{2n} & \text{or } y(xy)^{2n+1},\\[2mm]
 (b) & y(xy)^{2n} & \text{or } x(yx)^{2n+1},
\end{array}
\]
where \(n\in\mathbb{N}\). The elements of type (a), respectively of
type (b), are conjugate to \(x\), respectively to \(y\).\footnote{Translator's note: the
1970 text writes every element of \(G\) as \((xy)^n\), \(y(xy)^n\) or \((xy)^nx\) with
\(n\geq0\), and says that the second is conjugate to \(x\) and the third to \(y\); this
holds only for \(n\) odd (for \(n=0\) they are \(y\) and \(x\)). The corrected list is
that of the Gille--Polo re-edition (2011), Exposé~XXI, which also adds ``\(n>0\)'' to the
statement; we keep the 1970 statement, since the trivial subgroup, generated by
\((xy)^0=1\), is allowed. Either way, every element that is a word of odd length is
conjugate to \(x\) or to \(y\), so a normal subgroup containing neither lies in the
cyclic group generated by \(xy\), which gives the lemma.}
\end{proof}

\begin{remark}\label{sga3:XXI:3.4.5}
The integer \(n\) is computed immediately. Put
\[
 (\alpha^\ast,\beta)=p,\qquad
 (\beta^\ast,\alpha)=q.
\]
Then\footnote{Translator's note: the 1970 text has
\((s_\alpha s_\beta)(\alpha)=(pq-1)\alpha-p\beta\), \((s_\alpha s_\beta)(\beta)=q\alpha-\beta\)
and the matrix \(\left(\begin{smallmatrix}pq-1&-p\\ q&-1\end{smallmatrix}\right)\), with
\(p\) and \(q\) interchanged; the formulas are corrected in the Gille--Polo re-edition
(2011), Exposé~XXI. The order \(n\) is the same, both matrices having trace \(pq-2\) and
determinant \(1\).}
\[
 (s_\alpha s_\beta)(\alpha)=(pq-1)\alpha-q\beta,
 \qquad
 (s_\alpha s_\beta)(\beta)=p\alpha-\beta.
\]
Thus \(n\) is the order of the matrix
\[
 \begin{pmatrix}
  pq-1 & p\\
  -q   & -1
 \end{pmatrix}.
\]
One finds\footnote{Translator's note: the Gille--Polo re-edition (2011), Exposé~XXI, justifies the list. In our formulation: the
matrix has determinant \(1\) and trace \(pq-2\). By Corollary~\ref{sga3:XXI:2.2.2},
\(pq=0\) only if \(p=q=0\), and the matrix is then \(-1\); otherwise \(pq\in\{1,2,3\}\)
by Proposition~\ref{sga3:XXI:2.3.1}, the trace is \(-1\), \(0\) or \(1\), and the roots of
the characteristic polynomial \(t^2-(pq-2)t+1\) are primitive cube, fourth or sixth
roots of unity.}
\[
\begin{aligned}
 pq=0&\Longrightarrow n=2,\\
 pq=1&\Longrightarrow n=3,\\
 pq=2&\Longrightarrow n=4,\\
 pq=3&\Longrightarrow n=6.
\end{aligned}
\]

\smallskip
\noindent
\emph{N.B.} Expressing the finiteness of the order of the preceding
matrix recovers inequality \SGARefLink{sga3:XXI:eq:13}{(13)}.
\end{remark}
