
\begin{enumerate}
\item[\textup{(vii)}]\phantomsection\label{def:sga3-VIB-7.2-vii}\label{def:sga3-VIB-7.2.2}\label{def:sga3-VIB-7.2.2-a}%
  Let \(A\) and \(B\) be two geometrically reduced \(k\)-group
  subschemes of a \(k\)-group \(G\). The \emph{commutator group
  subscheme of \(A\) and \(B\) in \(G\)}, denoted
  \[
    (A,B)_G
    \qquad\text{or simply}\qquad
    (A,B),
  \]
  is the closed group subscheme of \(G\) generated by the
  morphism
  \phantomsection\label{form:sga3-VIB-7.2.2-commutator}%
  \[
    \nu:A\times_kB\longrightarrow G
  \]
  defined by
  \[
    (a,b)\longmapsto a\cdot b\cdot a^{-1}\cdot b^{-1}
    \qquad\text{for }a\in A(S)\text{ and }b\in B(S).
  \]
  When \(G\) is geometrically reduced over \(k\), the
  \emph{derived group of \(G\)}, denoted \(\mathfrak{D}(G)\), is
  the group \((G,G)\). For a geometrically reduced \(k\)-group
  \(G\) to be commutative, it is necessary and sufficient that
  \(\mathfrak{D}(G)\) be the unit \(k\)-group.

\item[\textup{(viii)}]\phantomsection\label{def:sga3-VIB-7.2-viii}
  We shall sometimes put
  \[
    \Gamma'_G(f)=\bigcup_{n\geq1}f^n(X^n)
  \]
  \textup{(}with the notation of
  \hyperref[concl:sga3-VIB-7.1-ii]{7.1\textup{(ii)}}\textup{)}. Notice
  that \(\Gamma'_G(f)\) is a subset of \(G\) stable under the group
  law, in the sense of \hyperref[par:sga3-VIB-3.0]{3.0}.
\end{enumerate}

\medskip
\phantomsection\label{cor:sga3-VIB-7.3}%
\noindent\textbf{Corollary 7.3.---}
\emph{Let \(G\) be a \(k\)-group, let \(X\) be a geometrically
reduced \(k\)-scheme, and let \(f:X\to G\) be a
\(k\)-morphism. Let \(S\) be a \(k\)-scheme.}
\begin{enumerate}
\item[\textup{\SGAInlineRefTarget{cor:sga3-VIB-7.3:item:i}(i)}]\phantomsection\label{concl:sga3-VIB-7.3-i}
  \emph{Let \(u\in\operatorname{End}_{S\text{-gr.}}(G_S)\) be such
  that one has the inclusion of presheaves}
  \[
    u\bigl(f_S(X_S)\bigr)\subset f_S(X_S)
  \]
  \emph{\textup{(}where \(f_S(X_S)\) denotes the subpresheaf of
  \(G_S\) image of \(X_S\) under \(f_S\)\textup{)}. Then}
  \[
    u\bigl((\Gamma_G(f))_S\bigr)\subset(\Gamma_G(f))_S
  \]
  \emph{\textup{(}as presheaves\textup{)}.}

\item[\textup{(i\,bis)}]\phantomsection\label{concl:sga3-VIB-7.3-ibis}
  \emph{Let \(u\in\operatorname{Aut}_{S\text{-gr.}}(G_S)\) be such
  that \(u(f_S(X_S))=f_S(X_S)\) \textup{(}as presheaves\textup{)};
  then \(u((\Gamma_G(f))_S)=(\Gamma_G(f))_S\) \textup{(}as
  presheaves\textup{)}.}

\item[\textup{\SGAInlineRefTarget{cor:sga3-VIB-7.3:item:ii}(ii)}]\phantomsection\label{concl:sga3-VIB-7.3-ii}
  \emph{Let \(u\in\operatorname{End}_{S\text{-gr.}}(G_S)\) be such
  that the restriction of \(u\) to the presheaf \(f_S(X_S)\) is the
  identity; then the restriction of \(u\) to the scheme
  \(\Gamma_G(f)_S\) is the identity automorphism.}

\item[\textup{\SGAInlineRefTarget{cor:sga3-VIB-7.3:item:iii}(iii)}]\phantomsection\label{concl:sga3-VIB-7.3-iii}
  \emph{Let \(Y\) be a geometrically reduced \(k\)-scheme and
  \(g:Y\to G\) a \(k\)-morphism. Suppose that, for every
  \(k\)-scheme \(S'\), every \(a\in X(S')\) and every
  \(b\in Y(S')\), the elements \(f\circ a\) and \(g\circ b\)
  commute \textup{(}respectively,
  \((\operatorname{int}(f\circ a))(g_{S'}(Y_{S'}))=g_{S'}(Y_{S'})\)\textup{)}.
  Then}
  \[
    \Gamma_G(f)\subset\underline{\operatorname{Centr}}_G\Gamma_G(g)
    \qquad\bigl(\text{respectively, }
    \Gamma_G(f)\subset\underline{\operatorname{Norm}}_G\Gamma_G(g)\bigr)
  \]
  \emph{\textup{(}as presheaves\textup{)}.}

\item[\textup{\SGAInlineRefTarget{cor:sga3-VIB-7.3:item:iv}(iv)}]\phantomsection\label{concl:sga3-VIB-7.3-iv}
  \emph{Let \(x\) be a point of \(G\) rational over \(k\). Then the
  \(k\)-group \(\Gamma_G(\{x\})\) is commutative.}

\item[\textup{(v)}]\phantomsection\label{concl:sga3-VIB-7.3-v}
  \emph{Let \(A\) and \(B\) be two group subschemes of \(G\),
  geometrically reduced over \(k\). If \(A\) and \(B\) are
  invariant (respectively, characteristic), then so is
  \((A,B)\).}
\end{enumerate}

We prove \textup{\SGARefLink{cor:sga3-VIB-7.3:item:i}{(i)}}. Put \(H=\Gamma_G(f)_S\) and
\(H'=u^{-1}(H)\). By \hyperref[concl:sga3-VIB-7.1-iii]{7.1\textup{(iii)}},
it is enough to show that \(f_S\) factors through \(H'\). By
hypothesis, \(u\circ f_S\) belongs to \(u(f_S(X_S))(X_S)\), hence
to \((f_S(X_S))(X_S)\), hence a fortiori to \(H(X_S)\); this means
that \(u\circ f_S\) factors through the canonical injection of
\(H\) into \(G_S\), and hence that \(f_S\) factors through \(H'\).

It is clear that \textup{\SGARefLink{concl:sga3-VIB-7.3-ibis}{(i\,bis)}} is a consequence of
\textup{\SGARefLink{cor:sga3-VIB-7.3:item:i}{(i)}}.

We prove \textup{\SGARefLink{cor:sga3-VIB-7.3:item:ii}{(ii)}}. Since \(G\) is separated over
\(k\) (\hyperref[sga3:VIA:0.3]{VI\(_A\), 0.3}),\footnote{Translator's note: the 1970 text refers here to
VI\textsubscript{A}, 0.2; the separation of group schemes over a field is VI\textsubscript{A}, 0.3, the
reference given in the Gille--Polo re-edition (2011), Exposé~VI\textsubscript{B}.} \(G_S\) is separated over \(S\), so the unit section
of \(G_S\) is a closed immersion (\hyperref[prop:sga3-VIB-5.1]{5.1}); therefore
\[
  \operatorname{Ker}\bigl(\mu\circ(u,c)\bigr)
\]
is a closed group subscheme of \(G_S\) (\(\mu\), respectively
\(c\), denoting the multiplication, respectively the inversion, of
\(G_S\)). By hypothesis, the presheaf
\(\operatorname{Ker}(\mu\circ(u,c))\) contains the presheaf
\(f_S(X_S)\); hence \(\operatorname{Ker}(\mu\circ(u,c))\) contains
\(\Gamma_G(f)_S\) (\hyperref[concl:sga3-VIB-7.1-iii]{7.1\textup{(iii)}}).

We prove \textup{\SGARefLink{cor:sga3-VIB-7.3:item:iii}{(iii)}}. Recall
(\hyperref[prop:sga3-VIB-6.2]{6.2\textup{(iii)}}) that
\(\underline{\operatorname{Centr}}_G\Gamma_G(g)\) (respectively,
\(\underline{\operatorname{Norm}}_G\Gamma_G(g)\)) is representable
by a closed group subscheme of \(G\). By hypothesis, in view of
\textup{\SGARefLink{cor:sga3-VIB-7.3:item:ii}{(ii)}} (respectively, \textup{\SGARefLink{cor:sga3-VIB-7.3:item:i}{(i)}}),
\(\underline{\operatorname{Centr}}_G\Gamma_G(g)\) (respectively,
\(\underline{\operatorname{Norm}}_G\Gamma_G(g)\)) contains \(f(X)\)
as a presheaf; it therefore contains \(\Gamma_G(f)\).

Assertion \textup{\SGARefLink{cor:sga3-VIB-7.3:item:iv}{(iv)}} is a special case of
\textup{\SGARefLink{cor:sga3-VIB-7.3:item:iii}{(iii)}}, in which one takes for \(f\) and \(g\)
the injection into \(G\) of the reduced subscheme of \(G\) whose
underlying space is the closed point \(\{x\}\). This subscheme is
geometrically reduced, since \(\kappa(x)=k\).

Finally, we prove \textup{\SGARefLink{concl:sga3-VIB-7.3-v}{(v)}}. Let \(S\) be a \(k\)-scheme,
and let \(u\) be an inner automorphism (respectively, an
automorphism of \(S\)-groups) of \(G_S\). By hypothesis,
\(u(A_S)=A_S\) and \(u(B_S)=B_S\); one easily deduces from this,
with the notation of \hyperref[def:sga3-VIB-7.2-vii]{7.2\textup{(vii)}}, that
\phantomsection\label{form:sga3-VIB-7.3-v-commutator-image}%
\[
  u\bigl(\nu_S(A_S\times_SB_S)\bigr)=\nu_S(A_S\times_SB_S);
\]
hence, by \textup{\SGARefLink{cor:sga3-VIB-7.3:item:i}{(i)}}, one has \(u((A,B))=(A,B)\).

\medskip\noindent
\emph{Addition} (after the Gille--Polo re-edition (2011), Exposé~VI\textsubscript{B}; our
formulation). The re-edition precedes 7.3 by two reminders and restates it with hypotheses on
underlying sets. Keep the notation of 7.3, and put \(\Gamma_S=\Gamma_G(f)_S\).

(a) The image presheaf \(f_S(X_S)\) assigns to an \(S\)-scheme \(T\) the set of the \(f_S\circ x\),
\(x\in X_S(T)\). If \(Z\) is a subscheme of \(G_S\), then \(f_S(X_S)\) is a subpresheaf of \(Z\) if and only
if \(f_S\) factors through \(Z\) (take \(x=\operatorname{id}_{X_S}\)); since \(X_S\) is reduced, \(X\)
being geometrically reduced, this is the case as soon as \(f_S(X_S)\subset Z\) set-theoretically. By
\hyperref[prop:sga3-VIB-6.2]{6.2}, \(\underline{\operatorname{Centr}}_G\Gamma_G(f)\) and
\(\underline{\operatorname{Norm}}_G\Gamma_G(f)\) are represented by closed group subschemes of \(G\).

(b) In (i) it is enough that \(u(f_S(X_S))\subset\Gamma_S\) set-theoretically: then \(u\circ f_S\) factors
through \(\Gamma_S\) by (a), so \(f_S\) factors through the closed group subscheme \(u^{-1}(\Gamma_S)\),
which therefore contains \(\Gamma_S\) by 7.1 (iii). Hence in (i\,bis) it is enough that \(u(f_S(X_S))\) and
\(u^{-1}(f_S(X_S))\) be contained in \(\Gamma_S\) set-theoretically, for instance that
\(u(f_S(X_S))=f_S(X_S)\) set-theoretically; and the hypothesis of (ii) means that \(u\circ f_S=f_S\).
Taking \(u=\operatorname{int}(h)\) for \(h\in G(S)\): \(h\) normalizes \(\Gamma_S\) if
\(\operatorname{int}(h)\) and \(\operatorname{int}(h^{-1})\) map \(f_S(X_S)\) into \(\Gamma_S\)
set-theoretically, and \(h\) centralizes \(\Gamma_S\) if \(\operatorname{int}(h)\circ f_S=f_S\).

(c) Consequently, a \(k\)-group subscheme \(H\) of \(G\) normalizes (respectively, centralizes)
\(\Gamma_G(f)\) if and only if, for every \(k\)-scheme \(S\) and every \(h\in H(S)\),
\(\operatorname{int}(h)\) maps \(f_S(X_S)\) into \(\Gamma_S\) set-theoretically (respectively,
\(\operatorname{int}(h)\circ f_S=f_S\)): the conditions are necessary because \(f_S\) factors through
\(\Gamma_S\), and they are sufficient by (b), since \(h^{-1}\in H(S)\).

(d) Likewise, the conclusion of (iii) concerning normalizers holds as soon as, for every \(k\)-scheme
\(S'\), every \(a\in X(S')\) and every \(b\in Y(S')\), the elements
\(\operatorname{int}(f\circ a)(g\circ b)\) and \(\operatorname{int}(f\circ a)^{-1}(g\circ b)\) of
\(G(S')\) belong to \(\Gamma_G(g)(S')\): by (b), applied over \(S'\) to \(u=\operatorname{int}(f\circ a)\),
\(f\circ a\) normalizes \(\Gamma_G(g)_{S'}\), and one concludes with (a) and 7.1 (i). The re-edition
asks this of the first element only, which does not suffice. Let \(G\) be the constant \(k\)-group
defined by the semidirect product \(\mathbb Z[1/2]\rtimes\mathbb Z\), in which \(n\in\mathbb Z\) acts on
\(\mathbb Z[1/2]\) by multiplication by \(2^n\); let \(g\) be the inclusion of the constant subgroup
\(Y=\mathbb Z_k\) of \(\mathbb Z[1/2]_k\), so that \(\Gamma_G(g)=Y\), and let
\(f\colon\operatorname{Spec}k\to G\) be the point \(t=(0,1)\). Then \(\operatorname{int}(t)\) maps
\(b\in Y\) to \(2b\in Y\); but \(\Gamma_G(f)\) contains \(t^{-1}\), and \(\operatorname{int}(t^{-1})\) maps
\(1\in Y(k)\) to \(1/2\notin Y(k)\), so \(\Gamma_G(f)\) does not normalize \(\Gamma_G(g)\).
