\phantomsection\label{prop:sga3-VIB-8.2}%
\noindent\textbf{Proposition 8.2.---}
\emph{Let \(\mathcal C\) be a category, let \(T\) be a
topology on \(\mathcal C\), let \(G\) be a sheaf of groups on
\(\mathcal C\), and let \(n\) be an integer with \(n\geq0\).
The following conditions are equivalent:}
\begin{enumerate}
\item[\textup{\SGAInlineRefTarget{prop:sga3-VIB-8.2:item:i}(i)}]\phantomsection\label{concl:sga3-VIB-8.2-i}
  \emph{Put \(K_0=G\), and, for \(p\geq1\), put}
  \[
    K_p=\operatorname{Comm}(K_{p-1},K_{p-1})
    \qquad
    \bigl(\textup{respectively, }\
      K_p=\operatorname{Comm}(G,K_{p-1})\bigr).
  \]
  \emph{Then \(K_n\) is the unit presheaf of groups.}

\item[\textup{\SGAInlineRefTarget{prop:sga3-VIB-8.2:item:ii}(ii)}]\phantomsection\label{concl:sga3-VIB-8.2-ii}
  \emph{Put \(K'_0=G\), and, for \(p\geq1\), put}
  \[
    K'_p=\operatorname{Comm}_T(K'_{p-1},K'_{p-1})
    \qquad
    \bigl(\textup{respectively, }\
      K'_p=\operatorname{Comm}_T(G,K'_{p-1})\bigr).
  \]
  \emph{Then \(K'_n\) is the unit sheaf of groups.}

\item[\textup{\SGAInlineRefTarget{prop:sga3-VIB-8.2:item:iii}(iii)}]\phantomsection\label{concl:sga3-VIB-8.2-iii}
  \emph{There exists a sequence \(H_0=G,H_1,\ldots,H_n\) of
  invariant subsheaves of \(G\) such that, for every \(i\), the
  quotient sheaf \(H_i/H_{i+1}\) is commutative
  \textup{(respectively, central in \(G/H_{i+1}\))}, and \(H_n\)
  is the unit sheaf.}
\end{enumerate}

It is clear that \(K_n\subset K'_n\); consequently,
\textup{\SGARefLink{prop:sga3-VIB-8.2:item:ii}{(ii)}} implies \textup{\SGARefLink{prop:sga3-VIB-8.2:item:i}{(i)}}. We show that
\textup{\SGARefLink{prop:sga3-VIB-8.2:item:i}{(i)}} implies \textup{\SGARefLink{prop:sga3-VIB-8.2:item:ii}{(ii)}}. It follows from
Expos\'e~IV, \SGARefLink{sga3:iv:remark:4:4:8}{4.4.7} that, if one denotes by
\(\overline A\) the sheaf associated with a presheaf \(A\), and if
\(A\) and \(B\) are two group subpresheaves of \(G\), then
\[
  \operatorname{Comm}_T(\overline A,\overline B)
  \subset\overline{\operatorname{Comm}(A,B)}.
\]
One deduces easily by induction that
\(K'_p\subset\overline{K_p}\), hence
\(K'_n\subset\overline{K_n}\); consequently, if \(K_n\) is the
unit presheaf, \(\overline{K_n}=K_n\) is the unit sheaf, and
therefore \(K'_n\) is the unit sheaf.

Finally, conditions \textup{\SGARefLink{prop:sga3-VIB-8.2:item:ii}{(ii)}} and \textup{\SGARefLink{prop:sga3-VIB-8.2:item:iii}{(iii)}} are
equivalent by the remark of \hyperref[par:sga3-VIB-8.1]{8.1}.

When these conditions are satisfied, the sheaf \(G\) is said
to be \emph{solvable of class \(n\)}
\textup{(}respectively, \emph{nilpotent of class \(n\)}\textup{)}.
When there exists an integer \(n\) such that these conditions are
satisfied, \(G\) is said to be \emph{solvable}
\textup{(}respectively, \emph{nilpotent}\textup{)}. Note that this
condition is independent of \(T\).\footnote{Translator's note: indeed condition
\textup{\SGARefLink{prop:sga3-VIB-8.2:item:i}{(i)}} of 8.2 does not involve \(T\), as the Gille--Polo re-edition (2011),
Exposé~VI\textsubscript{B}, points out.}

\phantomsection\label{prop:sga3-VIB-8.3}%
\noindent\textbf{Proposition 8.3.---}
\emph{Let \(k\) be a field, let \(S\) be a nonempty
\(k\)-scheme, let \(\Omega\) be an algebraically closed
extension of \(k\), and let \(G\) be a smooth \(k\)-group of
finite type. The following conditions are equivalent:}
\begin{enumerate}
\item[\textup{\SGAInlineRefTarget{prop:sga3-VIB-8.3:item:i}(i)}]\phantomsection\label{concl:sga3-VIB-8.3-i}
  \emph{\(G\) is solvable of class \(n\)
  \textup{(respectively, nilpotent of class \(n\))}.}

\item[\textup{(ii)}]\phantomsection\label{concl:sga3-VIB-8.3-ii}
  \emph{\(G\times_kS\) is solvable of class \(n\)
  \textup{(respectively, nilpotent of class \(n\))}.}

\item[\textup{(iii)}]\phantomsection\label{concl:sga3-VIB-8.3-iii}
  \emph{The group \(G(\Omega)\) is solvable of class \(n\)
  \textup{(respectively, nilpotent of class \(n\))}.}

\item[\textup{(iv)}]\phantomsection\label{concl:sga3-VIB-8.3-iv}
  \emph{Put \(K_0=G\), and, for \(p\geq1\), consider the
  \(k\)-groups}
  \[
    K_p=(K_{p-1},K_{p-1})
    \qquad
    \bigl(\textup{respectively, }\ K_p=(G,K_{p-1})\bigr)
  \]
  \emph{\textup{(cf.\ \hyperref[prop:sga3-VIB-7.8]{7.8})}.
  Then \(K_n\) is the unit \(k\)-group.}
\end{enumerate}

The equivalence of conditions \textup{\SGARefLink{prop:sga3-VIB-8.3:item:i}{(i)}} and \textup{\SGARefLink{concl:sga3-VIB-8.3-ii}{(ii)}}
follows from the definition
\textup{(}\hyperref[prop:sga3-VIB-8.2]{8.2}\textup{)}, since the
formation of the group presheaf of commutators commutes with
base change (IV, \SGARefLink{sga3:iv:proposition:4:1:3}{4.1.3}).

The equivalence of conditions \textup{\SGARefLink{prop:sga3-VIB-8.3:item:i}{(i)}} and \textup{\SGARefLink{concl:sga3-VIB-8.3-iv}{(iv)}} follows
from \hyperref[prop:sga3-VIB-7.8]{7.8}.\footnote{Translator's note: as the Gille--Polo re-edition (2011),
Exposé~VI\textsubscript{B}, explains, one shows by induction on \(p\) that each \(K_p\) is a smooth \(k\)-group
of finite type, invariant in \(G\) (7.1~(i), 7.3~(v), 7.8), and that it represents the sheaf
\(\operatorname{Comm}_T(K_{p-1},K_{p-1})\) \textup{(}respectively,
\(\operatorname{Comm}_T(G,K_{p-1})\)\textup{)}, \(T\) being the \((\mathrm{fppf})\) or \((\mathrm{fpqc})\)
topology, by 7.8 in case a); so the \(K_p\) of \textup{(iv)} represent the sheaves \(K'_p\) of 8.2~(ii).}

To prove the equivalence of conditions \textup{\SGARefLink{concl:sga3-VIB-8.3-iii}{(iii)}} and
\textup{\SGARefLink{concl:sga3-VIB-8.3-iv}{(iv)}}, we may assume that \(k=\Omega\); the result then
follows from \hyperref[cor:sga3-VIB-7.10]{7.10}.
