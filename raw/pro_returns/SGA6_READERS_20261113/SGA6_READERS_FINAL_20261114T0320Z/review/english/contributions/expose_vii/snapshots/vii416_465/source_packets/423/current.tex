Par suite:
\begin{equation}\tag{1.7.2}
\begin{aligned}
 i\ \text{régulière} &\Longleftrightarrow \forall i\geq1,
 \Tor_i^{A_0}(B_0,A)=0;\\
 i\circ j\ \text{régulière} &\Longleftrightarrow \forall i\geq1,
 \Tor_i^{A_0}(C_0,A)=0;\\
 j\ \text{régulière} &\Longleftrightarrow \forall i\geq1,
 \Tor_i^{B_0}(C_0,B)=0.
\end{aligned}
\end{equation}

\end{equation}
