
\textbf{Corollaire 6.8. (Grothendieck).} \emph{Soient $X$ projectif sur $S$ noeth\'erien, ${O}_X(1)$ tr\`es ample pour $X/S$, et $\mathcal J$ une famille de classes de faisceaux coh\'erents sur les fibres de $X/S$. On suppose}~:
\begin{enumerate}[label=(\alph*)]
\item \emph{Il existe un faisceau coh\'erent $E$ sur $X$ tel que $\mathcal J$ soit contenue dans la famille des classes de quotients de faisceaux de la forme $E_K$.}
\item[$(b_q)$] \emph{Dans les polyn\^omes de Hilbert $\chi(F_K(n))$ des $F_K\in\mathcal J$, les coefficients en degr\'es $\ge q-1$ restent major\'es.}
\end{enumerate}
\emph{Alors les $(F_K)_q$ (lorsque $F_K$ reste dans $\mathcal J$) forment une famille limit\'ee ; de plus les coefficients dans $\chi(F_K(n))$ en degr\'es $\ge q-2$ restent minor\'es.}

En effet, on peut \'evidemment supposer $X=\mathbb P^N_S$ et $E={O}_X(-m)^{\oplus M}$. Alors la premi\`ere assertion r\'esulte formellement de 6.7 (ii) et 1.13 ; la deuxi\`eme r\'esulte en raisonnant par r\'ecurrence \`a l'aide d'une suite exacte $0\to F(-1)\to F\to G\to0$, de 6.4 (ii).

\textbf{D\'efinition 6.9.} \emph{Soit $k$ un corps. On appelle \emph{$k$-cycle positif sp\'ecial de dimension $r$} tout $k$-sch\'ema projectif $X$, muni d'un faisceau ${O}_X(1)$ tr\`es ample, qui est r\'eunion de sous-sch\'emas ferm\'es $X_j$ de dimension $r$, $X_j$ \'etant obtenu par un changement de base $\operatorname{Spec}(k)\to\operatorname{Spec}(k_j)$ comme image inverse d'un $k_j$-sch\'ema \emph{int\`egre} $X'_j$ (muni de ${O}_{X'_j}(1)$). On appelle \emph{degr\'e} de $X$ le coefficient $a_r$ dans le polyn\^ome de Hilbert $\chi({O}_X(n))=\sum a_i\binom{n+i}{i}$.}

\textbf{Lemme 6.10.} \emph{Soient $k$ un corps alg\'ebriquement clos et $X$ un $k$-cycle positif sp\'ecial de dimension $r$ et degr\'e $d$. Alors ${O}_X$ est un $(0,\ldots,0,d)$-faisceau.}
