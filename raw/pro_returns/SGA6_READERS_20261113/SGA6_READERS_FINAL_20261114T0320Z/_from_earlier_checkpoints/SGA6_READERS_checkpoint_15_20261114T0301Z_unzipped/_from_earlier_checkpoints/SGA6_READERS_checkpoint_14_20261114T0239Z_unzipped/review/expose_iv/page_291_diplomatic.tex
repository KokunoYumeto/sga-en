B\arrow[r] & B'\\

\[
 B\otimes_A^{\mathrm L}A'\xrightarrow{\sim}B',

\[
 E\otimes_A^{\mathrm L}A'\xrightarrow{\sim}E\otimes_B^{\mathrm L}B',
\]
